\makeatletter  % Включаем режим чтения системных переменных Beamer
\setbeamertemplate{frametitle}{%
	\nointerlineskip%
	\hspace*{-\beamer@leftmargin}% Точный сдвиг к абсолютному левому краю слайда
	\begin{tikzpicture}[inner sep=0pt, outer sep=0pt]
		% 1. Горизонтальный градиент шапки ровно во всю ширину экрана
		\shade[left color=Burgundy, right color=BurgundyMid] (0,0) rectangle (\paperwidth,0.9);
		
		% 2. Текст заголовка слайда (красивый фиксированный отступ)
		\node[anchor=west, text=Ivory, align=left] at (0.6, 0.45) {%
			\usebeamerfont{frametitle}\insertframetitle%
		};
	\end{tikzpicture}\par%
	\nointerlineskip%
	\hspace*{-\beamer@leftmargin}% Точный сдвиг тени к левому краю
	\begin{tikzpicture}[inner sep=0pt, outer sep=0pt]
		% 3. Ослабленная, мягкая тень (смешиваем с Ivory для идеального размытия)
		\shade[top color=black!10!Ivory, bottom color=Ivory] (0,0) rectangle (\paperwidth,0.06);
	\end{tikzpicture}\par%
	\vspace*{-1ex}% Корректируем микропробел перед текстом слайда
}
\makeatother   % Выключаем режим чтения системных переменных
